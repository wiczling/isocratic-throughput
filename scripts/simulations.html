Introductions

A set of simulations exploring possible models and its modification. We illustrate it using the Neue model:

\[ logk(\varphi) = logk_w - S1 \cdot (1+S2) \cdot \varphi \cdot /(1+S2\cdot \varphi) \]

Combined. Dewetting and pH effect model

Code 
# Parameters
pHo      <- 3.2 
alpha  <- 3
pKa      <- 5.5 
logkw1  <- 1
logkw2  <- 2
S1 <- 4
S2 <- 2
logS2 = log10(S2)
fid <-  0.05
delta = 0.01
logks <- 0.5
pKas <- 6

pKasmpH = pKas-pHo
pHmpKa = pHo-pKa


dlogkw  <- logkw2-logkw1

funlogki_fi <- function(logkw1, S1, logks, S2,
                        pKasmpH, alpha, pHmpKa, dlogkw, fi) {
  
  log10 <- 2.302585092994046  # ln(10)
  
  logfix <- -(1 + S2) * S1 * fi / (1 + S2 * fi)
  
  t1 <- (pKasmpH - alpha * fi) * log10
  t2 <- (pHmpKa  + alpha * fi) * log10
  
  logksfix <- logks - log1p_exp(t1) / log10
  
  a <- (logkw1 - dlogkw + logfix) * log10
  b <- logksfix * log10
  c <- (logkw1 + logfix) * log10 + t2
  
  logk <- log_sum_exp(a, log_sum_exp(b, c)) / log10 -
    log1p_exp(t2) / log10
  
  return(logk)
}

funlogki_pKa <- function(logkw, S1, logks, logS2, fid,
                         pKasmpH, alpha, pHmpKa, dlogkw, fi) {
  
  S2 <- 10^logS2
  delta <- 0.1
  
logkd <- funlogki_fi(
    logkw, S1, logks, S2,
    pKasmpH, alpha, pHmpKa, dlogkw, fid
  )
  
  logk_raw <- funlogki_fi(
    logkw, S1, logks, S2,
    pKasmpH, alpha, pHmpKa, dlogkw, fi
  )
  
  w <- plogis((fid - fi) / 0.01)
  
  logk <- w * logkd + (1 - w) * logk_raw
}

# Function definitions
f_exact <- function(fi) {
  z <-  (pHo + alpha*fi)-(pKa)
  k1 = 10^(logkw1 - S1*(1+S2)*fi/(1+S2*fi))
  k2 = 10^(logkw2 - S1*(1+S2)*fi/(1+S2*fi))
  six = 1/(1+10^(pKas-(pHo+alpha*fi)))
  ks = 10^(logks)
  k1s = k1+ks*six
  logk = log10((k1s + k2 * 10^(z)) / (1 + 10^(z)))
}

f_1 <- function(fi) {
  k1 = 10^(logkw1 - S1*(1+S2)*fi/(1+S2*fi))
  six = 1/(1+10^(pKas-(pHo+alpha*fi)))
  ks = 10^(logks)
  logk = log10(k1+ks*six)
}

f_2 <- function(fi) {
  logk = logkw2 - S1*(1+S2)*fi/(1+S2*fi)
}

f_dewetting <- function(fi) {
 logkd = f_exact(fid)
 logk = f_exact(fi)
  w <- plogis((fid - fi) / 0.01)
  return(w * logkd + (1 - w) * logk)
}

funlogki_pKa_fi <- function(fi_vals) funlogki_pKa(logkw2, S1, logks, logS2, fid,
               pKasmpH, alpha, pHmpKa, dlogkw, fi_vals)


fi_vals <- seq(0, 1, by = 0.01)
df <- data.frame(

Combined. Dewetting and pH effect model (empirical)

$$ \[\begin{aligned} logk_1 &= logk_{w1} + c_\gamma \varphi - \dfrac{S_1(1+S_2)\varphi}{1+S_2\varphi}, \\[8pt] logk_2 &= logk_{w2} - \dfrac{S_1(1+S_2)\varphi}{1+S_2\varphi}, \\[8pt] logk_t &= \log_{10}\!\left( \frac{k_1 + k_2\,10^{(\left(pH_o + \alpha \varphi \right) - pK_a)}}{1 + 10^{(\left(pH_o + \alpha \varphi \right) - pK_a)}} \right),\\ logk_a &= logk_t(\varphi_{id}), \\ logk_b &= logk_t(\varphi), \\ w &= \frac{1}{1 + \exp\!\left(-\dfrac{\varphi_{id}-\varphi}{0.001}\right)}, \\ logk &= w\,logk_a + (1-w)\,logk_b. \end{aligned}\]

$$

Utylity function

\[ \begin{aligned} d &= \left| \log k_1 - \log k_2 \right|, \\ l_o &= 10^{\min\left(\log k_1,\ \log k_2\right)}, \\ h_i &= 10^{\max\left(\log k_1,\ \log k_2\right)}, \\ U &= \frac{1}{1 + e^{-(d - 0.07918125)/0.015}} \times \frac{1}{1 + e^{-(l_o - 2)/0.25}} \times \frac{1}{1 + e^{-(10 - h_i)/1.5}}. \end{aligned} \]

Code 
logk1 = seq(0, log10(14), length.out = 50)
logk2 = seq(0, log10(14), length.out = 50)
df <- expand.grid(logk1 = logk1, logk2=logk2)  %>%
 mutate(
    d  = abs(logk1 - logk2),
    lo = 10^(pmin(logk1, logk2)),
    hi = 10^(pmax(logk1, logk2)),
    utylity =
      plogis((d  - 0.07918125) / 0.015) *  # midpoint for selectivity of 1.2, zero for 1.04 and 1 for 1.5
      plogis((lo - 2)      / 0.25)  *
      plogis((10 - hi)     / 1.5)) %>%
  select(-d,-lo,-hi)

fig2<-ggplot(df, aes(x = 10^logk1, y = 10^logk2, z = utylity)) +
  geom_contour_filled()+
   scale_fill_grey(start = 0.9, end = 0.1) +
  labs(x = "Retention factor of analyte 1", y = "Retention factor of analyte 2")+

```{r combined-dewetting-pH}

funlogki_fi <- function(logkw1, S1, logks, S2,
                        pKasmpH, alpha, pHmpKa, dlogkw, fi) {
  
  log10 <- 2.302585092994046  # ln(10)
  
  logfix <- -(1 + S2) * S1 * fi / (1 + S2 * fi)
  
  t1 <- (pKasmpH - alpha * fi) * log10
  t2 <- (pHmpKa  + alpha * fi) * log10
  
  logksfix <- logks - log1p_exp(t1) / log10
  
  a <- (logkw1 - dlogkw + logfix) * log10
  b <- logksfix * log10
  c <- (logkw1 + logfix) * log10 + t2
  
  logk <- log_sum_exp(a, log_sum_exp(b, c)) / log10 -
    log1p_exp(t2) / log10
  
  return(logk)
}

funlogki_pKa <- function(logkw, S1, logks, logS2, fid,
                         pKasmpH, alpha, pHmpKa, dlogkw, fi) {
  
  S2 <- 10^logS2
  delta <- 0.1
  
logkd <- funlogki_fi(
    logkw, S1, logks, S2,
    pKasmpH, alpha, pHmpKa, dlogkw, fid
  )
  
  logk_raw <- funlogki_fi(
    logkw, S1, logks, S2,
    pKasmpH, alpha, pHmpKa, dlogkw, fi
  )
  
  w <- plogis((fid - fi) / 0.01)
  
  logk <- w * logkd + (1 - w) * logk_raw
}

# Function definitions
f_exact <- function(fi) {
  z <-  (pHo + alpha*fi)-(pKa)
  k1 = 10^(logkw1 - S1*(1+S2)*fi/(1+S2*fi))
  k2 = 10^(logkw2 - S1*(1+S2)*fi/(1+S2*fi))
  six = 1/(1+10^(pKas-(pHo+alpha*fi)))
  ks = 10^(logks)

  logk = log10((k1s + k2 * 10^(z)) / (1 + 10^(z)))
}
```

# Combined. Dewetting and pH effect model (empirical)
$$
\begin{aligned}
logk_1 &= logk_{w1} + c_\gamma \varphi
- \dfrac{S_1(1+S_2)\varphi}{1+S_2\varphi}, \\[8pt]
logk_2 &= logk_{w2}
- \dfrac{S_1(1+S_2)\varphi}{1+S_2\varphi}, \\[8pt]
logk_t &= \log_{10}\!\left(
\frac{k_1 + k_2\,10^{(\left(pH_o + \alpha \varphi \right) - pK_a)}}{1 + 10^{(\left(pH_o + \alpha \varphi \right) - pK_a)}}
\right),\\

logk_a &= logk_t(\varphi_{id}), \\
logk_b &= logk_t(\varphi), \\
w &= \frac{1}{1 + \exp\!\left(-\dfrac{\varphi_{id}-\varphi}{0.001}\right)}, \\
logk &= w\,logk_a + (1-w)\,logk_b.
\end{aligned}
$$

# Utylity function
$$
\begin{aligned}
d &= \left| \log k_1 - \log k_2 \right|, \\
l_o &= 10^{\min\left(\log k_1,\ \log k_2\right)}, \\
h_i &= 10^{\max\left(\log k_1,\ \log k_2\right)}, \\
U &=
\frac{1}{1 + e^{-(d - 0.07918125)/0.015}}
\times
\frac{1}{1 + e^{-(l_o - 2)/0.25}}
\times
\frac{1}{1 + e^{-(10 - h_i)/1.5}}.
\end{aligned}
$$
```{r utylity-function}
logk1 = seq(0, log10(14), length.out = 50)
logk2 = seq(0, log10(14), length.out = 50)
df <- expand.grid(logk1 = logk1, logk2=logk2)  %>%
 mutate(
    d  = abs(logk1 - logk2),
    lo = 10^(pmin(logk1, logk2)),
    hi = 10^(pmax(logk1, logk2)),
    utylity =
      plogis((d  - 0.07918125) / 0.015) *  # midpoint for selectivity of 1.2, zero for 1.04 and 1 for 1.5
      plogis((lo - 2)      / 0.25)  *
      plogis((10 - hi)     / 1.5)) %>%
  select(-d,-lo,-hi)

fig2<-ggplot(df, aes(x = 10^logk1, y = 10^logk2, z = utylity)) +
  geom_contour_filled()+

  labs(x = "Retention factor of analyte 1", y = "Retention factor of analyte 2")+
```
