Load data

data <-read.csv(here(data_deriv_dir, "data.csv"), header = TRUE)%>%
  mutate(Compound = HSM3_Name) %>%
  mutate(fi = phi/100,
         logk = log10(k))%>%
  filter(!is.na(logk)) %>%
  group_by(fi, Compound, Column_Phase) %>%
  summarise(logk = mean(logk)) %>%
  ungroup()%>%
   mutate(ID = match(Compound, unique(Compound)),
         column = match(Column_Phase, unique(Column_Phase)))
# for correlation plots
data_wide <- data %>%
  group_by(ID, fi, Compound, Column_Phase) %>%
  summarise(logk = mean(logk)) %>%
   pivot_wider(names_from = Column_Phase, values_from = logk)

analytes_names = data%>%distinct(ID, .keep_all = T)%>%select(ID, Compound)
column_names = data%>%distinct(column, .keep_all = T)%>%select(column,Column_Phase)

smiles <- read.csv(here(data_deriv_dir,"smiles.csv"), sep = ",", header = TRUE)%>%
  select(-ID)%>%
  left_join(analytes_names)%>%arrange(ID)%>%distinct(ID, .keep_all = T)

fg_df = read.csv(file = here(data_deriv_dir,"smarts_functional_groups.csv"), header = TRUE)%>%
  select(-id1,-id2) %>%
  rename(smiles=SMILES) %>%
  left_join(smiles,)%>%arrange(ID)%>%distinct(ID, .keep_all = T)

descriptor_df <- read.csv(file = here(data_deriv_dir,"descriptor_df.csv"), header = TRUE)%>%
  rename(smiles=SMILES) %>%
  left_join(smiles) %>%arrange(ID)%>%distinct(ID, .keep_all = T)

column_df = read.csv(file = here(data_deriv_dir,"columns.csv"), header = TRUE)%>%
  mutate(Column_Phase = str_remove(Phase, "Waters|Agilent"))%>%
  mutate(Column_Phase = str_remove(Column_Phase, " "))%>%
  mutate(Column_Phase = case_when(Column_Phase == "Varian/C18-A" ~ "C18-A",
                                  Column_Phase == "CSH Phenyl- Hexyl"~"CSH Phenyl-Hexyl",
                                 .default = Column_Phase)) %>%
  left_join(column_names) %>%
  rename("carbon_loading" = "Carbon.Loading....",
         "Endcapping" = "Endcapping...."
         ) %>%
  arrange(column)


analytes_df = read.csv(file = here(data_deriv_dir,"analytes.csv"), header = TRUE)

Initial estimates

Code 
init_aprox <- data %>%
  group_by(Compound, Column_Phase) %>%
  summarise(
    S1 = if_else(n() == 1,-16,polyfit(fi / (1 + 2 * fi), logk, 1)[1]),
    logkw = if_else(n() == 1,polyfit(-16 * fi / (1 + 2 * fi), logk, 0), 
                             polyfit(fi / (1 + 2 * fi), logk, 1)[2]),
    S2 = 2) %>%
  ungroup() %>%
  select(Compound, Column_Phase, logkw, S2, S1) %>%
  mutate(S1 = -S1 / (1 + 2))

init_aprox = init_aprox %>%left_join(descriptor_df%>%select(Compound,MolWt,LogP))
